# 深層礦脈｜信箱與開發者權限 v0.9.7

> 這份解決兩件事：
> 1. 朋友有帳號之後，**開發者模式不能給他們用**
> 2. 你要能**發信給全體玩家、附獎勵**

---

## 1. 開發者模式怎麼鎖的

玩家版**根本不會下載 `editor.js`**。要叫出編輯模式，兩個條件同時成立才行：

1. 網址後面要帶 `?dev=1`
2. 而且 —— **在你自己電腦上直接開檔案**（`file://` 或 `localhost`），**或** 你的雲端帳號在 `admins` 名單裡

所以：

| 情況 | 結果 |
|---|---|
| 你在家用 Chrome 開 `index.html?dev=1` | ✎ 出現（本機一律放行，方便你開發） |
| 你在手機上 `https://...github.io/?dev=1`，已用你的帳號登入 | ✎ 出現 |
| 朋友在手機上打 `?dev=1`（沒在 admins 名單） | ✎ **不出現**，editor.js 也不會被下載 |
| 誰都沒加 `?dev=1` | ✎ 不出現 |

**老實說**：`editor.js` 這個檔案還是放在 GitHub 上，真的懂的人能自己叫起來。這擋的是「朋友手滑按到把數值改爛」，不是擋駭客。

## 2. 建資料表（貼一次就好）

Supabase → **SQL Editor** → **New query** → 整段貼上 → **Run**。

```sql
-- === 管理員名單 ===
create table if not exists public.admins (
  user_id uuid primary key references auth.users(id) on delete cascade,
  note    text
);
alter table public.admins enable row level security;
drop policy if exists "see self admin" on public.admins;
create policy "see self admin" on public.admins for select using (auth.uid() = user_id);

-- === 信件 ===
create table if not exists public.mail (
  id         bigint generated always as identity primary key,
  title      text not null,
  body       text not null default '',
  coins      bigint not null default 0,
  ore_name   text,
  ore_qty    int not null default 0,
  tool_id    text,
  tool_qty   int not null default 0,
  to_user    uuid references auth.users(id) on delete cascade,  -- null = 全體
  created_at timestamptz default now(),
  expires_at timestamptz
);
alter table public.mail enable row level security;
drop policy if exists "read my mail"  on public.mail;
drop policy if exists "admin writes"  on public.mail;
create policy "read my mail" on public.mail for select
  using (to_user is null or to_user = auth.uid());
create policy "admin writes" on public.mail for all
  using (exists (select 1 from public.admins a where a.user_id = auth.uid()))
  with check (exists (select 1 from public.admins a where a.user_id = auth.uid()));

-- === 領取記錄（主鍵擋住重複領） ===
create table if not exists public.mail_claims (
  user_id    uuid   references auth.users(id) on delete cascade,
  mail_id    bigint references public.mail(id) on delete cascade,
  claimed_at timestamptz default now(),
  primary key (user_id, mail_id)
);
alter table public.mail_claims enable row level security;
drop policy if exists "my claims read"   on public.mail_claims;
drop policy if exists "my claims insert" on public.mail_claims;
create policy "my claims read"   on public.mail_claims for select using (auth.uid() = user_id);
create policy "my claims insert" on public.mail_claims for insert with check (auth.uid() = user_id);
```

## 3. 發信指令（這段是重點）

再貼一次下面整段，它會建立四個**一行就能用**的指令。建好之後你**永遠不用再寫 SQL**。

```sql
-- 發全體信
create or replace function public.send_mail(
  title text, body text default '', coins bigint default 0,
  ore text default null, ore_qty int default 0,
  tool text default null, tool_qty int default 0,
  days int default 30
) returns bigint language plpgsql security invoker as $$
declare new_id bigint;
begin
  if not exists (select 1 from public.admins where user_id = auth.uid()) then
    raise exception '你不是管理員，不能發信';
  end if;
  if coalesce(trim(title), '') = '' then raise exception '標題不能空白'; end if;
  if coins < 0 or coins > 1000000 then raise exception '金幣要在 0 ~ 1,000,000 之間（你填了 %）', coins; end if;
  insert into public.mail (title, body, coins, ore_name, ore_qty, tool_id, tool_qty, to_user, expires_at)
  values (title, body, coins, ore, ore_qty, tool, tool_qty, null, now() + (days || ' days')::interval)
  returning id into new_id;
  return new_id;
end $$;

-- 發給指定一個人（用他註冊的 Email）
create or replace function public.send_to(
  email text, title text, body text default '', coins bigint default 0,
  ore text default null, ore_qty int default 0,
  tool text default null, tool_qty int default 0,
  days int default 30
) returns bigint language plpgsql security definer set search_path = public as $$
declare uid uuid; new_id bigint;
begin
  if not exists (select 1 from public.admins where user_id = auth.uid()) then
    raise exception '你不是管理員，不能發信';
  end if;
  select id into uid from auth.users u where u.email = send_to.email;
  if uid is null then raise exception '找不到這個 Email 的玩家：%', email; end if;
  if coins < 0 or coins > 1000000 then raise exception '金幣要在 0 ~ 1,000,000 之間'; end if;
  insert into public.mail (title, body, coins, ore_name, ore_qty, tool_id, tool_qty, to_user, expires_at)
  values (title, body, coins, ore, ore_qty, tool, tool_qty, uid, now() + (days || ' days')::interval)
  returning id into new_id;
  return new_id;
end $$;

-- 收回一封信（發錯了用這個）
create or replace function public.unsend_mail(mail_id bigint)
returns text language plpgsql security invoker as $$
declare n int;
begin
  if not exists (select 1 from public.admins where user_id = auth.uid()) then
    raise exception '你不是管理員';
  end if;
  select count(*) into n from public.mail_claims c where c.mail_id = unsend_mail.mail_id;
  delete from public.mail m where m.id = unsend_mail.mail_id;
  return format('已收回第 %s 號信，已經有 %s 個人領過了（領過的不會被收回去）', mail_id, n);
end $$;

-- 看發過哪些信、幾個人領了
create or replace function public.list_mail()
returns table (id bigint, title text, 獎勵 text, 對象 text, 已領 bigint, 發送時間 timestamptz)
language sql security invoker as $$
  select m.id, m.title,
         concat_ws(' ', nullif('$' || m.coins, '$0'),
                        nullif(m.ore_name || ' x' || m.ore_qty, null),
                        nullif(m.tool_id || ' x' || m.tool_qty, null)),
         case when m.to_user is null then '全體' else '指定個人' end,
         (select count(*) from public.mail_claims c where c.mail_id = m.id),
         m.created_at
  from public.mail m order by m.id desc limit 50;
$$;
```

### 怎麼用（複製改字就好）

到 **SQL Editor**，貼一行，按 Run：

```sql
-- 純公告，不給獎勵
select send_mail('版本更新 v0.9.7', '新增了信箱，之後有活動會從這裡發獎勵。');

-- 給全體 5000 金幣
select send_mail('開幕慶', '感謝大家幫忙試玩！', 5000);

-- 給全體 5000 金幣 + 一把鐵鎬
select send_mail('開幕慶', '感謝大家幫忙試玩！', 5000, null, 0, 'iron', 1);

-- 單獨補償某個人
select send_to('朋友的@email.com', '存檔補償', '上次卡住很抱歉。', 20000);

-- 看發過什麼
select * from list_mail();

-- 發錯了，收回第 3 號信
select unsend_mail(3);
```

### 為什麼搞砸不了

- 這些指令**只碰 `mail` 這張表**，碰不到玩家存檔。最慘的狀況就是多發一封信，`unsend_mail` 收回。
- 打錯字會直接跳中文錯誤（「標題不能空白」「找不到這個 Email 的玩家」），**不會半做半不做**。
- 金幣超過 100 萬會被擋下來，手滑多打兩個 0 不會出事。
- 不是管理員執行 → 直接被拒絕。

### 先拿自己試

第一封信建議先發給自己確認流程：

```sql
select send_to('notzutote0148@gmail.com', '測試', '這是測試信', 100);
```

進遊戲按右上角 ✉ 看得到、領得到，再發全體的。

## 4. 遊戲裡長什麼樣

- 右上角多一個 **✉**，有沒領的信會冒紅點和數字
- 點開看到信件列表：標題、內文、附件獎勵、領取按鈕
- 領過的顯示「已領取」，過期的顯示「已過期」
- 沒登入雲端 → 信箱是空的，並提示要先登入

**領取流程**：先去雲端搶下「領取記錄」，搶到了才在本機加獎勵，然後存檔上傳。
所以同一封信**重複點、兩支手機同時點，都只會領到一次**（第二次會跳「這封信已經領過了」）。

---

## 5. 一樣要講清楚的限制

獎勵是**玩家的瀏覽器自己加上去的**，伺服器只記「誰領過」。
所以有人改前端，還是能給自己加金幣——這跟信箱無關，是整個遊戲在客戶端跑的必然結果。
真正要防，得把遊戲邏輯搬到伺服器，那是第三台之後的事。
